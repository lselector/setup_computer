# .zshrc - interactive zsh only.
# Mirrors the interactive part of ~/.bashrc.
# Environment lives in ~/.zshenv.
# ------------------------------------
# Re-apply env: macOS /etc/zprofile (path_helper)
# moved system dirs ahead of ours in login shells.
source ~/.zshenv
# ------------------------------------ prompt
# Same as bash: ~/D/G/seminar/work (master)$
setopt PROMPT_SUBST
precmd () {
    PS1_PATH=$(sed "s:\([^/\.]\)[^/]*/:\1/:g" <<< ${PWD/#$HOME/\~})
    PS1_GIT=$(git branch --show-current 2>/dev/null)
}
PROMPT='$PS1_PATH${PS1_GIT:+ ($PS1_GIT)}%(!.#.$) '
# ------------------------------------ history
HISTFILE=$HOME/.zsh_history
HISTSIZE=10000
SAVEHIST=100000
# ------------------------------------
# function to source file only if it exists
mysource () {
    [[ -f "$1" ]] && source "$1"
}
# ------------------------------------
mysource $HOME/.myaliases # aliases for bash and fish
mysource $HOME/.mygit
autoload -Uz compinit && compinit   # after .mygit sets fpath
mysource $HOME/.myvv
