# ------------------------------------
# .bashrc for local machine only (then jumps into fish)
# ------------------------------------
export SHELL=/bin/bash
export BASH_SILENCE_DEPRECATION_WARNING=1
export PYTHONDONTWRITEBYTECODE=1
export VIRTUAL_ENV_DISABLE_PROMPT=1
export CLAUDE_CODE_SHELL=/bin/bash   # Claude Code tools run in bash
export PROMPT_COMMAND='PS1_PATH=$(sed "s:\([^/\.]\)[^/]*/:\1/:g" <<< ${PWD/#$HOME/\~})'
export PS1='$PS1_PATH\$ '
# ------------------------------------
PATH=$PATH:.
PATH=$PATH:$HOME/docs/bin
PATH=$PATH:/opt/homebrew/opt/ffmpeg-full/bin
PATH=/opt/homebrew/bin:$PATH:/usr/local/bin
PATH=$PATH:/usr/bin
PATH=$PATH:/bin
PATH=$PATH:/usr/sbin
PATH=$PATH:/sbin
PATH="$HOME/.local/bin:$PATH"     # for claude CLI
export PATH

PYTHONPATH=.:./py_lib:$HOME/docs/py_lib
PYTHONPATH=$PYTHONPATH:$HOME/Documents/GitHub/setup_computer/py_lib
export PYTHONPATH
# ------------------------------------
export SSHDIR=$HOME/.ssh
export HISTSIZE=10000
export HISTFILESIZE=100000
# ------------------------------------
# function to source file only if it exists
mysource () {
    [[ -f "$1" ]] && source "$1"
}
# ------------------------------------
mysource $HOME/.myaliases # aliases for bash and fish
mysource $HOME/.mygit
mysource $HOME/.myvv
# ------------------------------------
mysource $HOME/.venvs/standard/bin/activate
# ------------------------------------
# Keep this block at the end of .bashrc.
# It loads the bash env above, then replaces bash with fish
# only if: the shell is interactive ('i' in $-), fish exists,
# this is the first bash in the session (FISH_FLAG), and we
# are not inside Claude Code (CLAUDECODE=1).
FISH_LOCATION=/opt/homebrew/bin/fish

if [ -z "$FISH_FLAG" ]; then
    export FISH_FLAG="not_set"
else
    export FISH_FLAG="was_set"
fi

if echo $- | grep -q 'i' && [[ -x $FISH_LOCATION ]] && [[ "$FISH_FLAG" = "not_set" ]] && [[ -z "$CLAUDECODE" ]]; then
    exec env SHELL=$FISH_LOCATION $FISH_LOCATION -i
fi
