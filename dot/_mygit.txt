# git completion for the running shell
if [ -n "$ZSH_VERSION" ]; then
    [ -f ~/.git-completion.zsh ] && . ~/.git-completion.zsh
elif [ -n "$BASH_VERSION" ]; then
    [ -f ~/.git-completion.bash ] && . ~/.git-completion.bash
fi
