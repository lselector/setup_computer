# Remap Google Drive path to ~/docs
GDRIVE_EMAIL="your.name@gmail.com"
GDRIVE_PATH1="$HOME/My Drive ($GDRIVE_EMAIL)"
GDRIVE_PATH2="$HOME/Library/CloudStorage"
GDRIVE_PATH2+="/GoogleDrive-$GDRIVE_EMAIL"
GDRIVE_PATH2+="/My Drive"
DOCS_PATH="$HOME/docs"
case "$PWD" in
    "$GDRIVE_PATH1"*)
        NEW_DIR="${PWD/$GDRIVE_PATH1/$DOCS_PATH}"
        cd "$NEW_DIR" 2>/dev/null
        ;;
    "$GDRIVE_PATH2"*)
        NEW_DIR="${PWD/$GDRIVE_PATH2/$DOCS_PATH}"
        cd "$NEW_DIR" 2>/dev/null
        ;;
esac
unset GDRIVE_EMAIL GDRIVE_PATH1 GDRIVE_PATH2 DOCS_PATH NEW_DIR
